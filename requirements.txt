# Pinned to versions with no known advisories as of 2026-11-11, checked with
# pip-audit. The previous pin set carried 23 known vulnerabilities across
# requests, urllib3, lxml and starlette.
#
# starlette 1.x is what forces Python 3.11: the advisories affecting 0.38.x are
# only fixed in releases that require it. Staying on 3.9 would mean shipping
# known-vulnerable versions, so the floor moved instead.
fastapi==0.162.4
uvicorn==0.54.0
requests==2.34.2
urllib3==2.8.0
beautifulsoup4==4.15.0
lxml==6.1.3
defusedxml==0.7.1

# Transitive, but listed explicitly with a floor rather than left to resolve.
# certifi is the CA bundle every HTTPS verification depends on, and pip will
# happily leave an old one in place when a loose constraint upstream is still
# satisfied: an existing environment was found holding 2026.7.22, which
# carries PYSEC-2026-4320, after an otherwise clean upgrade. A floor rather
# than a pin, because a CA bundle should keep moving forward.
certifi>=2026.10.18
# soupsieve is beautifulsoup4's CSS selector engine, and the same floating
# problem reached it: an environment was found on 2.9.2 carrying three
# advisories while beautifulsoup4's own constraint was still satisfied.
soupsieve>=4.0.0
